# 1. Import Libraries
 
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
 
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
 
 
# 2. Load Dataset
 
df = pd.read_csv("BostonHousing.csv")
 
print(df.head())
print("Dataset Shape:", df.shape)
 
 
# 3. Preprocessing
 
df = df.loc[:, ~df.columns.str.contains("^Unnamed")]
 
target_col = next(
    col for col in df.columns
    if col.lower() == "medv"
)
 
X = df.drop(columns=[target_col])
y = df[target_col]
 
X = X.drop(columns=["CAT.MEDV"], errors="ignore")
 
 
# 4. Split Dataset (70% Training, 30% Testing)
 
X_train, X_test, y_train, y_test = train_test_split(
    X, y,
    test_size=0.30,
    random_state=42
)
 
print("Training Shape:", X_train.shape)
print("Testing Shape:", X_test.shape)
 
 
# 5. Standardize Features
 
scaler = StandardScaler()
 
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)
 
 
# 6. Train Linear Regression Model
 
model = LinearRegression()
 
model.fit(X_train_scaled, y_train)
 
print("Model Trained Successfully!")
 
 
# 7. Predict Training and Testing Data
 
y_train_pred = model.predict(X_train_scaled)
y_test_pred = model.predict(X_test_scaled)
 
 
# 8. Calculate MSE and R2 Score
 
train_mse = mean_squared_error(y_train, y_train_pred)
test_mse = mean_squared_error(y_test, y_test_pred)
 
train_r2 = r2_score(y_train, y_train_pred)
test_r2 = r2_score(y_test, y_test_pred)
 
print("\nTraining Results")
print("----------------------")
print("Training MSE:", train_mse)
print("Training R2 Score:", train_r2)
 
print("\nTesting Results")
print("----------------------")
print("Testing MSE:", test_mse)
print("Testing R2 Score:", test_r2)
 
 
# 9. Plot Training Error vs Testing Error
 
plt.figure(figsize=(8, 5))
 
plt.bar(
    ["Training Error", "Testing Error"],
    [train_mse, test_mse]
)
 
plt.title("Training Error vs Testing Error")
plt.xlabel("Dataset")
plt.ylabel("Mean Squared Error")
 
plt.show()
 
 
# 10. Plot Actual vs Predicted Values
 
plt.figure(figsize=(8, 5))
 
plt.scatter(y_test, y_test_pred)
 
min_val = min(y_test.min(), y_test_pred.min())
max_val = max(y_test.max(), y_test_pred.max())
 
plt.plot(
    [min_val, max_val],
    [min_val, max_val],
    linestyle="--"
)
 
plt.title("Actual vs Predicted House Prices")
plt.xlabel("Actual House Prices")
plt.ylabel("Predicted House Prices")
 
plt.show()

    accuracy_score,
    confusion_matrix,
    classification_report,
    precision_score,
    recall_score,
    f1_score,
    roc_curve,
    roc_auc_score
)

data = load_breast_cancer()

# 3. Handle Missing Values

# 1. Import Libraries
 
import numpy as np

# 4. Apply K-Means Clustering
 
kmeans = KMeans(
    n_clusters=3,
    random_state=42,
    n_init=10
)
 
clusters = kmeans.fit_predict(X_scaled)
 
print("\nCluster Labels:")

X_pca = pca.fit_transform(X_scaled)

    X_pca[:, 0],
    X_pca[:, 1],
    c=clusters,
    cmap="viridis",
    s=50
)
 
plt.title("K-Means Clustering (PCA 2D)")
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
 
plt.colorbar(label="Cluster")
plt.show()

plt.figure(figsize=(8, 6))
 
plt.scatter(
    X_pca[:, 0],
    X_pca[:, 1],
    c=y,
    cmap="viridis",

plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")

comparison = pd.DataFrame({

})
 
print("\nActual Labels vs Predicted Clusters:")
print(comparison.head(20))

print("\nCluster Centers:")
print(kmeans.cluster_centers_)

# 1. Import Libraries
 
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
 
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
 
from sklearn.metrics import (
    accuracy_score,
    confusion_matrix,
    classification_report,
    precision_score,
    recall_score,
    f1_score
)
 
 
# 2. Load Dataset
 
df = pd.read_csv("spam.csv", encoding="latin-1")
 
df = df.iloc[:, :2]
df.columns = ["label", "message"]
 
print(df.head())
print(df.shape)
 
 
# 3. Preprocessing
 
df.dropna(inplace=True)
 
df["label"] = df["label"].map({
    "ham": 0,
    "spam": 1
})
 
 
# 4. Split Dataset
 
X_train, X_test, y_train, y_test = train_test_split(
    df["message"],
    df["label"],
    test_size=0.30,
    random_state=42,
    stratify=df["label"]
)
 
 
# 5. Convert Text into TF-IDF Features
 
vectorizer = TfidfVectorizer(stop_words="english")
 
X_train = vectorizer.fit_transform(X_train)
X_test = vectorizer.transform(X_test)
 
 
# 6. Train Naive Bayes Model
 
nb_model = MultinomialNB()
 
nb_model.fit(X_train, y_train)
 
 
# 7. Prediction
 
y_pred_nb = nb_model.predict(X_test)
 
 
# 8. Evaluation
 
print("\nNaive Bayes Results")
 
print("Accuracy:", accuracy_score(y_test, y_pred_nb))
 
print("Precision:", precision_score(y_test, y_pred_nb))
 
print("Recall:", recall_score(y_test, y_pred_nb))
 
print("F1 Score:", f1_score(y_test, y_pred_nb))
 
cm_nb = confusion_matrix(y_test, y_pred_nb)
 
print("\nConfusion Matrix:")
print(cm_nb)
 
print("\nClassification Report:")
print(classification_report(y_test, y_pred_nb))
 
 
# 9. Visualize Confusion Matrix
 
plt.figure(figsize=(6, 5))
 
sns.heatmap(
    cm_nb,
    annot=True,
    fmt="d",
    cmap="Blues",
    xticklabels=["Ham", "Spam"],
    yticklabels=["Ham", "Spam"]
)
 
plt.title("Naive Bayes Confusion Matrix")
plt.xlabel("Predicted")
plt.ylabel("Actual")
 
plt.show()
 
 
# 10. Visualize Class-Conditional Word Probabilities
 
feature_names = vectorizer.get_feature_names_out()
 
spam_prob = nb_model.feature_log_prob_[1]
ham_prob = nb_model.feature_log_prob_[0]
 
top_spam = np.argsort(spam_prob)[-10:]
top_ham = np.argsort(ham_prob)[-10:]
 
plt.figure(figsize=(10, 5))
 
plt.bar(
    feature_names[top_spam],
    np.exp(spam_prob[top_spam])

import numpy as np
import random
 
# Environment setup
GRID_SIZE = 6
START = (0, 0)
GOAL = (5, 5)
ACTIONS = ["Up", "Down", "Left", "Right"]
 
# Create 36 states
states = [(i, j) for i in range(GRID_SIZE) for j in range(GRID_SIZE)]
 
# Randomly place 5 obstacles, excluding start and goal
random.seed(42)
available = [s for s in states if s != START and s != GOAL]
OBSTACLES = set(random.sample(available, 5))
 
print("Obstacles:", OBSTACLES)
 
# Initialize Q-table
Q = np.zeros((36, 4))
 
# Hyperparameters
alpha = 0.1
gamma = 0.9
epsilon = 1.0
epsilon_min = 0.01
epsilon_decay = 0.995
episodes = 5000
max_steps = 100
 
def state_to_index(state):
    return state[0] * GRID_SIZE + state[1]
 
def get_next_state(state, action):
    row, col = state
 
    if action == 0:      # Up
        row -= 1
    elif action == 1:    # Down
        row += 1
    elif action == 2:    # Left
        col -= 1
    elif action == 3:    # Right
        col += 1
 
    # Prevent movement outside grid
    row = max(0, min(GRID_SIZE - 1, row))
    col = max(0, min(GRID_SIZE - 1, col))
 
    return (row, col)
 
# Q-learning training
for episode in range(episodes):
    state = START
 
    for step in range(max_steps):
 
        state_index = state_to_index(state)
 
        # Epsilon-greedy action selection
        if random.random() < epsilon:
            action = random.randint(0, 3)
        else:
            action = np.argmax(Q[state_index])
 
        next_state = get_next_state(state, action)
 
        # Calculate reward
        if next_state in OBSTACLES:
            reward = -5
            next_state = state
            done = False
        elif next_state == GOAL:
            reward = 10
            done = True
        else:
            reward = -1
            done = False
 
        next_index = state_to_index(next_state)
 
        # Q-learning update
        Q[state_index, action] += alpha * (
            reward + gamma * np.max(Q[next_index]) - Q[state_index, action]
        )
 
        state = next_state
 
        if done:
            break
 
    epsilon = max(epsilon_min, epsilon * epsilon_decay)
 
# Print final Q-table
print("\nFinal Q-table:")
print(np.round(Q, 2))
 
# Display learned policy
symbols = ["↑", "↓", "←", "→"]
print("\nLearned Policy:")
 
for i in range(GRID_SIZE):
    for j in range(GRID_SIZE):
        state = (i, j)
 
        if state == START:
            print("S", end="  ")
        elif state == GOAL:
            print("G", end="  ")
        elif state in OBSTACLES:
            print("X", end="  ")
        else:
            best_action = np.argmax(Q[state_to_index(state)])
            print(symbols[best_action], end="  ")
 
    print()
 
# Find optimal path using learned Q-table
state = START
path = [state]
visited = {state}
 
for _ in range(100):
    if state == GOAL:
        break
 
    state_index = state_to_index(state)
    sorted_actions = np.argsort(Q[state_index])[::-1]
 
    moved = False
 
    for action in sorted_actions:
        next_state = get_next_state(state, action)
 
        if next_state not in OBSTACLES and next_state not in visited:
            state = next_state
            path.append(state)
            visited.add(state)
            moved = True
            break
 
    if not moved:
        break
 
print("\nOptimal Path:")
print(path)
 
if path[-1] == GOAL:
    print("Goal reached successfully!")
else:
    print("Goal not reached. Try increasing training episodes.")

import gymnasium as gym

# Hyperparameters
episodes = 1000
alpha = 0.1
epsilon = 1.0
epsilon_min = 0.01
epsilon_decay = 0.995
 
gamma_values = [0.5, 0.9, 0.99]
 
# Store results for different gamma values
all_rewards = {}
all_q_tables = {}
 
# Train agent for each gamma value
for gamma in gamma_values:
 
    env = gym.make("Taxi-v3")
 
    state_size = env.observation_space.n
    action_size = env.action_space.n
 
    # Initialize Q-table
    Q = np.zeros((state_size, action_size))
 
    epsilon = 1.0
    episode_rewards = []
 
    # Training
    for episode in range(episodes):
 
        state, info = env.reset()
        total_reward = 0
        done = False
 
        while not done:
 
            # Epsilon-greedy action selection
            if random.random() < epsilon:
                action = env.action_space.sample()
            else:
                action = np.argmax(Q[state])
 
            # Take action
            next_state, reward, terminated, truncated, info = env.step(action)
 
            done = terminated or truncated
 
            # Q-learning update
            if done:
                target = reward
            else:
                target = reward + gamma * np.max(Q[next_state])
 
            Q[state, action] += alpha * (target - Q[state, action])
 
            state = next_state
            total_reward += reward
 
        episode_rewards.append(total_reward)
 
        # Reduce exploration
        epsilon = max(epsilon_min, epsilon * epsilon_decay)
 
    all_rewards[gamma] = episode_rewards
    all_q_tables[gamma] = Q
 
    env.close()
 
    print(f"Training completed for gamma = {gamma}")
 
 
# Plot episode reward vs number of episodes

def make_sequences(frame, feature_cols, seq_len):
    """
    Make sequences separately per city.
    Sequence rows: [t-seq_len, ..., t-1]
    Target: AQI at time t.
    """
    X_list, y_list, city_list, time_list = [], [], [], []
 
    for city, group in frame.groupby(CITY_COL):
        group = group.sort_values(TIME_COL).reset_index(drop=True)
        values = group[feature_cols].values.astype(np.float32)
        targets = group[TARGET_COL].values.astype(np.float32)
 
        for i in range(seq_len, len(group)):
            X_list.append(values[i-seq_len:i])
            y_list.append(targets[i])
            city_list.append(city)
            time_list.append(group.loc[i, TIME_COL])
 
    return (
        np.asarray(X_list, dtype=np.float32),
        np.asarray(y_list, dtype=np.float32),
        np.asarray(city_list),
        np.asarray(time_list)
    )
 
# Build sequences on each partition. To retain prior observations at split
# boundaries, build sequences from the full city history and select by target date.
def make_sequences_with_date_splits(frame, feature_cols, seq_len):
    X_list, y_list, city_list, time_list = [], [], [], []
    for city, group in frame.groupby(CITY_COL):
        group = group.sort_values(TIME_COL).reset_index(drop=True)
        values = group[feature_cols].values.astype(np.float32)
        targets = group[TARGET_COL].values.astype(np.float32)
        for i in range(seq_len, len(group)):
            X_list.append(values[i-seq_len:i])
            y_list.append(targets[i])
            city_list.append(city)
            time_list.append(group.loc[i, TIME_COL])
    return (np.asarray(X_list), np.asarray(y_list),
            np.asarray(city_list), pd.to_datetime(np.asarray(time_list)))

# ------------------------- DEEP LEARNING MODELS -------------------------
 
if RUN_DEEP_MODELS and HAS_TF and len(X_seq_train) > 0 and len(X_seq_test) > 0:
 
    def compile_and_fit(model, model_name):
        model.compile(
            optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
            loss="mse",
            metrics=["mae"]
        )
        callbacks = [
            tf.keras.callbacks.EarlyStopping(
                monitor="val_loss", patience=5,
                restore_best_weights=True
            )
        ]
 
        validation_data = (
            (X_seq_val, y_seq_val) if len(X_seq_val) else None
        )
 
        start = time.perf_counter()
        model.fit(
            X_seq_train, y_seq_train,
            validation_data=validation_data,
            epochs=EPOCHS,
            batch_size=BATCH_SIZE,
            callbacks=callbacks,
            verbose=1
        )
        train_seconds = time.perf_counter() - start
 
        start = time.perf_counter()
        pred = model.predict(X_seq_test, batch_size=BATCH_SIZE, verbose=0).reshape(-1)
        inference_seconds = time.perf_counter() - start
 
        evaluate_predictions(
            model_name, y_seq_test, pred,
            train_seconds, inference_seconds,
            seq_test_cities, seq_test_times
        )
        return model, pred
 
    n_features = X_seq_train.shape[2]
 
    # 1) LSTM
    lstm = tf.keras.Sequential([
        layers.Input(shape=(SEQ_LEN, n_features)),
        layers.LSTM(64),
        layers.Dropout(0.2),
        layers.Dense(32, activation="relu"),
        layers.Dense(1)
    ])
    lstm, lstm_pred = compile_and_fit(lstm, "LSTM")
 
    # 2) BiLSTM with temporal attention
    inp = layers.Input(shape=(SEQ_LEN, n_features))
    h = layers.Bidirectional(layers.LSTM(64, return_sequences=True))(inp)
    score = layers.Dense(1, activation="tanh")(h)
    weights = layers.Softmax(axis=1, name="temporal_attention_weights")(score)
    context = layers.Lambda(
        lambda z: tf.reduce_sum(z[0] * z[1], axis=1)
    )([h, weights])
    h2 = layers.Dense(32, activation="relu")(context)
    h2 = layers.Dropout(0.2)(h2)
    out = layers.Dense(1)(h2)
    bilstm_attention = Model(inp, out)
    bilstm_attention, bilstm_pred = compile_and_fit(
        bilstm_attention, "BiLSTM + Attention"
    )
 
    # 3) 1D CNN regression
    cnn = tf.keras.Sequential([
        layers.Input(shape=(SEQ_LEN, n_features)),
        layers.Conv1D(64, kernel_size=3, padding="same", activation="relu"),
        layers.Conv1D(32, kernel_size=3, padding="same", activation="relu"),
        layers.GlobalAveragePooling1D(),
        layers.Dense(32, activation="relu"),
        layers.Dense(1)
    ])
    cnn, cnn_pred = compile_and_fit(cnn, "1D-CNN")
 
    # 4) Transformer Encoder regressor
    def transformer_block(x, head_size=32, num_heads=4, ff_dim=64, dropout=0.1):
        attn = layers.MultiHeadAttention(
            num_heads=num_heads, key_dim=head_size, dropout=dropout
        )(x, x)
        x = layers.LayerNormalization(epsilon=1e-6)(x + attn)
        ff = layers.Dense(ff_dim, activation="relu")(x)
        ff = layers.Dropout(dropout)(ff)
        ff = layers.Dense(x.shape[-1])(ff)
        return layers.LayerNormalization(epsilon=1e-6)(x + ff)
 
    inp = layers.Input(shape=(SEQ_LEN, n_features))
    x = layers.Dense(64)(inp)
    # Learned positional embedding
    positions = tf.range(start=0, limit=SEQ_LEN, delta=1)
    pos_embedding = layers.Embedding(input_dim=SEQ_LEN, output_dim=64)(positions)
    x = x + pos_embedding
    x = transformer_block(x)
    x = transformer_block(x)
    x = layers.GlobalAveragePooling1D()(x)
    x = layers.Dense(32, activation="relu")(x)
    out = layers.Dense(1)(x)
    transformer = Model(inp, out)
    transformer, transformer_pred = compile_and_fit(transformer, "Transformer")
 
    # 5) XGBoost + LSTM hybrid: concatenate XGBoost prediction and LSTM output,
    # then learn a linear fusion weight using validation predictions.
    if HAS_XGBOOST:
        # Create sequence-aligned tabular rows using the target date/city.
        seq_test_frame = pd.DataFrame({
            CITY_COL: seq_test_cities,
            TIME_COL: pd.to_datetime(seq_test_times)
        })
        test_rows = test_df.copy()
        test_rows[TIME_COL] = pd.to_datetime(test_rows[TIME_COL])
 
        aligned_test = seq_test_frame.merge(
            test_rows, on=[CITY_COL, TIME_COL], how="left"
        )
        valid_alignment = aligned_test[TARGET_COL].notna().values
 
        # Fit XGB on tabular training data, then get predictions for sequence dates.
        xgb_hybrid = XGBRegressor(
            n_estimators=400, max_depth=6, learning_rate=0.05,
            subsample=0.8, colsample_bytree=0.8,
            objective="reg:squarederror", random_state=RANDOM_STATE,
            n_jobs=-1
        )
        start = time.perf_counter()
        xgb_hybrid.fit(X_train_dense, y_train)
        xgb_train_seconds = time.perf_counter() - start
 
        # Tabular feature row matching each sequence target.
        xgb_test_aligned = preprocessor.transform(
            aligned_test.loc[valid_alignment, FEATURES + [CITY_COL]]
        )
        if hasattr(xgb_test_aligned, "toarray"):
            xgb_test_aligned = xgb_test_aligned.toarray()
 
        start = time.perf_counter()
        xgb_aligned_pred = xgb_hybrid.predict(xgb_test_aligned)
        xgb_inf_seconds = time.perf_counter() - start
 
        # LSTM output is aligned with sequence test target rows.
        lstm_aligned_pred = lstm_pred[valid_alignment]
        y_hybrid_true = y_seq_test[valid_alignment]
 
        # Learn blend weights on validation set when enough validation data exists.
        # For this simple example use a validation-set linear regression fusion.
        val_frame = pd.DataFrame({
            CITY_COL: seq_cities[seq_val_mask],
            TIME_COL: pd.to_datetime(np.asarray(seq_times)[seq_val_mask])
        }).merge(
            val_df, on=[CITY_COL, TIME_COL], how="left"
        )
        val_valid = val_frame[TARGET_COL].notna().values
 
        if len(X_seq_val) and val_valid.sum() > 0:
            lstm_val_pred = lstm.predict(X_seq_val, verbose=0).reshape(-1)
            xgb_val_matrix = preprocessor.transform(
                val_frame.loc[val_valid, FEATURES + [CITY_COL]]
            )
            if hasattr(xgb_val_matrix, "toarray"):
                xgb_val_matrix = xgb_val_matrix.toarray()
            xgb_val_pred = xgb_hybrid.predict(xgb_val_matrix)
            blend_model = LinearRegression()
            blend_model.fit(
                np.column_stack([
                    xgb_val_pred,
                    lstm_val_pred[val_valid]
                ]),
                val_frame.loc[val_valid, TARGET_COL].values
            )
            hybrid_pred = blend_model.predict(
                np.column_stack([xgb_aligned_pred, lstm_aligned_pred])
            )
        else:
            hybrid_pred = 0.5 * xgb_aligned_pred + 0.5 * lstm_aligned_pred
 
        evaluate_predictions(
            "XGBoost + LSTM",
            y_hybrid_true, hybrid_pred,
            xgb_train_seconds, xgb_inf_seconds,
            seq_test_cities[valid_alignment],
            seq_test_times[valid_alignment]
        )
