def evaluate_classifier(name, model, X_test, y_test):
    predictions = model.predict(X_test)
    probabilities = model.predict_proba(X_test)[:, 1]

    cm = confusion_matrix(y_test, predictions)

    results = {
        "Precision": precision_score(y_test, predictions, zero_division=0),
        "Recall": recall_score(y_test, predictions, zero_division=0),
        "F1 Score": f1_score(y_test, predictions, zero_division=0),
        "Balanced Accuracy": balanced_accuracy_score(y_test, predictions),
        "ROC-AUC": roc_auc_score(y_test, probabilities)
    }

    print(f"\n{name}")
    print("-"*50)
    print("Confusion Matrix:\n", cm)
    print("\nClassification Report:\n",
          classification_report(y_test, predictions, zero_division=0))
    print("Metrics:")
    for metric, value in results.items():
        print(f"{metric}: {value:.4f}")

    return results

# -------------------- (a) Environment setup --------------------

GRID_SIZE = 6

START = (0, 0)
GOAL = (5, 5)

OBSTACLES = {
    (1, 1),
    (1, 3),
    (2, 3),
    (3, 1),
    (4, 3),
    (4, 4)
}

# Actions: Up, Down, Left, Right
ACTIONS = {
    0: (-1, 0),
    1: (1, 0),
    2: (0, -1),
    3: (0, 1)
}

ACTION_NAMES = {
    0: "UP",
    1: "DOWN",
    2: "LEFT",
    3: "RIGHT"
}

N_STATES = GRID_SIZE * GRID_SIZE
N_ACTIONS = 4

def state_to_index(state):
    return state[0] * GRID_SIZE + state[1]

def index_to_state(index):
    return divmod(index, GRID_SIZE)

def step(state, action):

    dr, dc = ACTIONS[action]

    new_state = (
        state[0] + dr,
        state[1] + dc
    )

    # Prevent movement outside the grid.
    if not (
        0 <= new_state[0] < GRID_SIZE
        and 0 <= new_state[1] < GRID_SIZE
    ):
        return state, -1, False

    # Obstacle penalty.
    if new_state in OBSTACLES:
        return state, -8, False

    # Goal reward.
    if new_state == GOAL:
        return new_state, 25, True

    # Normal movement.
    return new_state, -1, False


# Display grid world
def display_grid():

    grid = np.zeros((GRID_SIZE, GRID_SIZE))

    for r, c in OBSTACLES:
        grid[r, c] = -1

    grid[START] = 2
    grid[GOAL] = 3

    plt.figure(figsize=(6, 6))
    plt.imshow(grid, cmap="coolwarm")

    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):

            if (r, c) == START:
                label = "START"
            elif (r, c) == GOAL:
                label = "GOAL"
            elif (r, c) in OBSTACLES:
                label = "X"
            else:
                label = ""

            plt.text(c, r, label, ha="center", va="center")

    plt.xticks(range(GRID_SIZE))
    plt.yticks(range(GRID_SIZE))
    plt.title("6x6 Grid World")
    plt.grid()
    plt.show()

display_grid()

# -------------------- (b) Q-Learning implementation --------------------

def train_q_learning(
    alpha=0.1,
    gamma=0.95,
    epsilon=1.0,
    epsilon_decay=0.995,
    min_epsilon=0.05,
    episodes=3000,
    seed=42
):

    rng = np.random.default_rng(seed)

    Q = np.zeros((N_STATES, N_ACTIONS))

    rewards_history = []
    epsilon_history = []

    for episode in range(episodes):

        state = START
        total_reward = 0

        for step_number in range(150):

            state_index = state_to_index(state)

            # Epsilon-greedy action selection.
            if rng.random() < epsilon:
                action = rng.integers(N_ACTIONS)
            else:
                action = np.argmax(Q[state_index])

            next_state, reward, done = step(state, action)

            next_index = state_to_index(next_state)

            # Q-learning update rule.
            if done:
                target = reward
            else:
                target = reward + gamma * np.max(Q[next_index])

            Q[state_index, action] += alpha * (
                target - Q[state_index, action]
            )

            total_reward += reward
            state = next_state

            if done:
                break

        rewards_history.append(total_reward)

        epsilon = max(
            min_epsilon,
            epsilon * epsilon_decay
        )

        epsilon_history.append(epsilon)

    return Q, rewards_history, epsilon_history

# -------------------- (d) Greedy policy evaluation --------------------

def evaluate_policy(Q):

    state = START
    path = [state]

    visited = {state}

    total_reward = 0
    reached_goal = False

    for _ in range(100):

        state_index = state_to_index(state)

        action = np.argmax(Q[state_index])

        next_state, reward, done = step(state, action)

        total_reward += reward

        state = next_state
        path.append(state)

        if done:
            reached_goal = True
            break

        # Protect against loops.
        if state in visited:
            print("Loop detected. Stopping evaluation.")
            break

        visited.add(state)

    return path, total_reward, reached_goal

    plt.text(

Q3 Critical Analysis:

1. Q-Learning:
   Q-learning is a model-free reinforcement learning algorithm.
   It learns the value of taking an action in a particular state.

2. Update Rule:
   Q(s,a) = Q(s,a) + alpha * [
       reward + gamma * max(Q(s',a')) - Q(s,a)
   ]

3. Epsilon-Greedy:
   With probability epsilon, a random action is selected.
   Otherwise, the action with the highest Q-value is selected.

4. Epsilon Decay:
   Epsilon gradually decreases during training.
   This allows exploration initially and exploitation later.

5. Learning Rate:
   A higher learning rate updates Q-values more aggressively.
   A lower learning rate produces smaller updates and may learn
   more gradually.

6. High Training Reward:
   A high reward during training does not guarantee a reliable
   policy because the agent may have learned a policy that works
   for frequently visited states but not other situations.

7. Tabular Q-Learning Limitation:
   The Q-table grows with the number of states and actions.
   Therefore, tabular Q-learning becomes inefficient for large or
   continuous state spaces.

8. Loop Prevention:
   A visited-state set prevents infinite loops during policy
   evaluation.
